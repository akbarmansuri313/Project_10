pipeline {
    agent any

    tools {
        maven 'Maven'
        jdk 'jdk-17'
    }

    stages {

        stage('Clean Workspace') {
            steps {
                deleteDir()
            }
        }

	

        stage('Clone Backend Repo') {
            steps {
                dir('backend') {
                    git branch: 'main',
                    url: 'https://github.com/akbarmansuri313/Project_10.git'
                }
            }
        }

        stage('Clone Frontend Repo') {
            steps {
                dir('frontend') {
                    git branch: 'main',
                    url: 'https://github.com/akbarmansuri313/Angular_10.git'
                }
            }
        }

        stage('Build Angular') {
            steps {
                dir('frontend/ORSPROJECT-10-UI') {
                    bat 'npm install'
                    bat 'npm run build -- --output-path=dist/app'
                }
            }
        }

        stage('Verify Angular Build') {
            steps {
                dir('frontend/ORSPROJECT-10-UI') {
                    bat 'dir dist\\app'
                }
            }
        }

       stage('Build Spring Boot') {
    steps {
        dir('backend/ORSProject-10') {
            bat 'mvn clean package -DskipTests'
        }
    }
}

        stage('Stop Old Containers') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10/docker-compose.yml down'
            }
        }

        stage('Build Docker Images') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10/docker-compose.yml build --no-cache'
            }
        }

        stage('Run Containers') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10/docker-compose.yml up -d'
            }
        }

        stage('Check Containers') {
            steps {
                bat 'docker ps'
            }
        }
    }
}
